\section{Optimization Problem}

\begin{equation}
\min_{\mathbf{u} \in \mathbb{R}^{N \times d}} \;
\mathcal{L}(\mathbf{u}) =
J_{\text{pose}} + J_{\text{bound}} + J_{\text{reg}} + J_{\text{target}}
+ C_{\text{scene}} + C_{\text{self}} + J_{\text{rel}}
\label{eq:full}
\end{equation}

where $N = 16$ is the number of B-spline knots (action horizon),
$d = 14$ is the total joint DOF (7 per arm).
$\mathbf{u} = [\mathbf{u}_1, \ldots, \mathbf{u}_N]^\top \in \mathbb{R}^{N \times d}$
is the sequence of B-spline control points, optimized by LBFGS.

Each term is structured as \textbf{weight $\times$ normalized error}:
$\text{term} = w \cdot e(\cdot)$, where $e(\cdot)$ is a differentiable, weight-independent
scalar measuring a specific type of violation or deviation. This decomposition separates
tuning (weights) from geometry (error metrics).

\section{Soft Costs}

\subsection{End-Effector Pose Tracking $J_{\text{pose}}$}

For each tool frame $f \in \mathcal{F}$, using the \textbf{axis-angle} method
(\texttt{use\_lie\_group: false}):

\begin{equation}
J_{\text{pose}} = w_p \cdot e_{\text{pos}} + w_r \cdot e_{\text{rot}}
\label{eq:pose}
\end{equation}

where the normalized error terms are:
%
\begin{align}
e_{\text{pos}} &= \sum_{f \in \mathcal{F}} \sum_{k=1}^{N}
\frac{1}{2} \bigl\lVert \mathbf{W}_p \odot (\mathbf{p}_{k,f} - \mathbf{p}_f^{\text{goal}}) \bigr\rVert^2
\label{eq:epos} \\
e_{\text{rot}} &= \sum_{f \in \mathcal{F}} \sum_{k=1}^{N}
\lVert\boldsymbol{\omega}_{k,f}\rVert^2
\label{eq:erot}
\end{align}

Here $\mathbf{W}_p = \operatorname{diag}(\hat{w}_x^p, \hat{w}_y^p, \hat{w}_z^p) \in \mathbb{R}^{3 \times 3}$
is the per-axis position weight matrix (defaults to identity),
$\boldsymbol{\omega}_{k,f}$ is the axis-angle representation of the relative rotation
$\mathbf{q}_{k,f} \otimes \bigl(\mathbf{q}_f^{\text{goal}}\bigr)^{-1}$:
%
\begin{align*}
\mathbf{q}_{\Delta} &= \mathbf{q}_{k,f} \otimes \bigl(\mathbf{q}_f^{\text{goal}}\bigr)^{-1} \\
\mathbf{v} &= \mathbf{W}_r \odot \operatorname{vec}(\mathbf{q}_{\Delta}) \\
\theta &= 2 \operatorname{arctan2}\bigl(\lVert\mathbf{v}\rVert,\; |w|\bigr) \\
\boldsymbol{\omega}_{k,f} &= \theta \cdot \frac{\mathbf{v}}{\lVert\mathbf{v}\rVert}
\end{align*}

The gradient decomposes multiplicatively:
%
\begin{equation}
\nabla_{\mathbf{p}} J_{\text{pose}} = w_p \cdot \nabla_{\mathbf{p}} e_{\text{pos}},
\qquad
\nabla_{\boldsymbol{\omega}} J_{\text{pose}} = w_r \cdot \nabla_{\boldsymbol{\omega}} e_{\text{rot}}
\end{equation}
where $\nabla_{\mathbf{p}} e_{\text{pos}} = \mathbf{W}_p^2 \odot (\mathbf{p} - \mathbf{p}^{\text{goal}})$,
$\nabla_{\boldsymbol{\omega}} e_{\text{rot}} = 2\,\boldsymbol{\omega}$.

\begin{quote}
\textbf{Note:} When \texttt{use\_lie\_group: true}, the Lie group logarithmic map
$\log: SO(3) \to \mathfrak{so}(3)$ replaces the axis-angle extraction with
a proper manifold-aware Jacobian.
\end{quote}

\subsection{Joint-Space Bound Penalty $J_{\text{bound}}$}

For each joint $d$ and state quantity
$s \in \{\text{pos}, \text{vel}, \text{acc}, \text{jerk}, \text{torque}\}$,
a hinge penalty activates when the state exceeds \textbf{activation-shrunken bounds}:

\begin{equation}
J_{\text{bound}} = \sum_{s} w_b^s \cdot e_{\text{bound}}^s
\label{eq:bound}
\end{equation}
%
\begin{equation}
e_{\text{bound}}^s = \sum_{k=1}^{N} \sum_{d=1}^{d}
\frac{1}{2} \Bigl(
\bigl[\max(0,\; x_{k,d}^s - \bar{u}_d^s)\bigr]^2
+ \bigl[\max(0,\; \bar{l}_d^s - x_{k,d}^s)\bigr]^2
\Bigr)
\label{eq:ebound}
\end{equation}

The shrunken bounds are ($\eta^s$ is the activation distance ratio):
%
\begin{equation}
\bar{l}_d^s = l_d^s + \eta^s (u_d^s - l_d^s), \qquad
\bar{u}_d^s = u_d^s - \eta^s (u_d^s - l_d^s)
\end{equation}

The activation distance $\eta^s$ creates a ``warning zone'' inside the hard limits,
making the cost gradient non-zero before the limit is actually reached. This prevents
the optimizer from running into the joint limit at full speed.

\subsection{L2 Smoothness Regularization $J_{\text{reg}}$}

Always-active Tikhonov regularization, independent of the bound penalty:

\begin{equation}
J_{\text{reg}} = \lambda_v \cdot e_{\text{vel}} + \lambda_a \cdot e_{\text{acc}} + \lambda_j \cdot e_{\text{jerk}}
\label{eq:reg}
\end{equation}
%
\begin{equation}
e_{\text{vel}} = \frac{1}{2} \sum_{k=1}^{N} \sum_{d=1}^{d} \dot{q}_{k,d}^2, \quad
e_{\text{acc}} = \frac{1}{2} \sum_{k=1}^{N} \sum_{d=1}^{d} \ddot{q}_{k,d}^2, \quad
e_{\text{jerk}} = \frac{1}{2} \sum_{k=1}^{N} \sum_{d=1}^{d} \dddot{q}_{k,d}^2
\label{eq:ereg}
\end{equation}

\texttt{retime\_regularization\_weights: true} means $\lambda$ is scaled by $dt^n$, so
the discrete sum approximates the continuous-time integral
$\int \lambda_a \lVert\ddot{\mathbf{q}}(t)\rVert^2 dt$, making the regularization
invariant to the choice of $dt$.

\subsection{Joint Target Distance $J_{\text{target}}$}

Pulls the trajectory toward the default joint configuration, with terminal weighting:

\begin{equation}
J_{\text{target}} = w_t \cdot e_{\text{target}}
\label{eq:target}
\end{equation}
%
\begin{equation}
e_{\text{target}} = \sum_{k=1}^{N} \alpha_k \sum_{d=1}^{d} (q_{k,d} - q_d^{\text{target}})^2
\label{eq:etarget}
\end{equation}
%
\begin{equation}
\alpha_k = \begin{cases}
1.0  & k = N \quad \text{(terminal knot)} \\
0.05 & k < N \quad \text{(non-terminal)}
\end{cases}
\end{equation}

The terminal bias ($\alpha_N = 1$, $\alpha_{k<N} = 0.05$) encodes that reaching the
goal configuration matters far more than the path taken to get there.
Gradient: $\nabla_{q_{k,d}} e_{\text{target}} = 2 \alpha_k (q_{k,d} - q_d^{\text{target}})$.

\subsection{Relative Pose Coordination $J_{\text{rel}}$ (runtime toggle)}

Simple continuous L2 cost, no deadzone, no layering — the form LBFGS handles best.

\begin{equation}
J_{\text{rel}} = w_{\text{rel},p} \cdot e_{\text{rel,pos}} + w_{\text{rel},r} \cdot e_{\text{rel,rot}}
\label{eq:rel}
\end{equation}
%
\begin{equation}
e_{\text{rel,pos}} = \sum_{k=1}^{N}
\frac{1}{2} \bigl\lVert \Delta\mathbf{p}_k - \Delta\mathbf{p}^{\text{target}} \bigr\rVert^2, \qquad
e_{\text{rel,rot}} = \sum_{k=1}^{N}
\bigl\lVert \boldsymbol{\omega}_k^{\text{rel}} \bigr\rVert^2
\label{eq:erel}
\end{equation}

The relative pose is computed in the primary's local frame ($\mathbf{q}^*$ denotes
quaternion conjugate):
%
\begin{align}
\Delta\mathbf{p}_k &= \mathbf{q}_{k,\text{primary}}^* \odot \bigl( \mathbf{p}_{k,\text{secondary}} - \mathbf{p}_{k,\text{primary}} \bigr) \\
\Delta\mathbf{q}_k &= \mathbf{q}_{k,\text{primary}}^* \otimes \mathbf{q}_{k,\text{secondary}}
\end{align}

\section{Collision Costs}

\subsection{Scene Collision Cost $C_{\text{scene}}$}

For each robot sphere $i$ at each timestep, using a \textbf{smooth activation function}
($C^1$-continuous, quadratic-linear transition):

\begin{equation}
C_{\text{scene}} = w_{\text{col}} \cdot e_{\text{scene}}, \qquad
e_{\text{scene}} = \sum_{k=1}^{N} \sum_{i=1}^{N_s} f_{\eta}\!\bigl(d_{k,i}^{\text{pen}}\bigr)
\label{eq:scene}
\end{equation}

The penetration depth is
$d_{k,i}^{\text{pen}} = r_i + \eta - \operatorname{sdf}(\mathbf{c}_i)$
($\operatorname{sdf}$ is negative inside obstacles).
The smooth activation function (source: \texttt{wp\_collision\_common.py:12-38}):

\begin{equation}
f_{\eta}(d) = \begin{cases}
0                        & d \le 0 \\[4pt]
\dfrac{d^2}{2\eta}       & 0 < d \le \eta \quad \text{(quadratic region)} \\[8pt]
d - \dfrac{\eta}{2}      & d > \eta \quad \text{(linear region)}
\end{cases}
\end{equation}

The function is $C^1$-continuous at $d = 0$ and $d = \eta$:
$f_{\eta}(0) = 0$, $f_{\eta}'(0) = 0$;
$f_{\eta}(\eta) = \eta/2$, $f_{\eta}'(\eta) = 1$.

The $C^1$ continuity is essential: $f_{\eta}'(0) = 0$ ensures the gradient vanishes
smoothly at the activation boundary, preventing discontinuous jumps in the gradient
field that would destabilize LBFGS. The quadratic-to-linear transition ($f_{\eta}'(0)=0$,
$f_{\eta}'(\eta)=1$) provides a ``soft landing'': near obstacles the cost rises quadratically
(gentle push), then transitions to linear (constant-strength push) when penetration is severe.

Sweep mode (\texttt{use\_sweep: true}, \texttt{use\_speed\_metric: true}) additionally
accounts for sphere velocity over $dt$ to compensate for the temporal discretization
gap between consecutive timesteps.

\subsection{Self-Collision Cost $C_{\text{self}}$}

For all predefined sphere pairs $(i,j)$, penalizes the maximally penetrating pair:

\begin{equation}
C_{\text{self}} = w_{\text{self}} \cdot e_{\text{self}}, \qquad
e_{\text{self}} = \sum_{k=1}^{N}
\max_{(i,j)} \Bigl[ \frac{1}{2} \bigl[ \max\bigl(0,\;
(r_i + r_j + \delta_{ij})^2 - \lVert\mathbf{c}_{k,i} - \mathbf{c}_{k,j}\rVert^2 \bigr) \bigr] \Bigr]
\label{eq:self}
\end{equation}

where $\delta_{ij}$ is pair-specific padding. The $\max_{(i,j)}$ operator selects only
the worst collision pair per timestep, producing a \textbf{sparse gradient}:
%
\begin{equation}
\nabla_{\mathbf{c}_i} e_{\text{self}} = -(\mathbf{c}_i - \mathbf{c}_j), \qquad
\nabla_{\mathbf{c}_j} e_{\text{self}} = (\mathbf{c}_i - \mathbf{c}_j)
\end{equation}
computed only for the maximally penetrating pair.

\section{Cost / Constraint / Convergence Taxonomy}

In cuRobo's rollout architecture, cost terms are classified into three categories:

\begin{center}
\begin{tabular}{@{}llll@{}}
\toprule
Category & Config Field & Behavior & Examples \\
\midrule
\textbf{Cost}       & \texttt{cost\_cfg}       & Smooth penalty, minimized by optimizer & Pose error, cspace distance, smoothness \\
\textbf{Constraint} & \texttt{constraint\_cfg} & Binary feasibility (\texttt{convert\_to\_binary: True}) & Collision, joint limits \\
\textbf{Convergence}& \texttt{convergence\_cfg} & Tolerance check for termination & Position/orientation tolerance \\
\bottomrule
\end{tabular}
\end{center}

Success condition = $\text{feasible} \land \text{converged}$.
Feasibility requires all constraint values $\le 0$.
Convergence requires all tolerance metrics below their thresholds.

\textbf{Important:} During MPC optimization, collision costs in both \texttt{cost\_cfg}
and \texttt{constraint\_cfg} use \textbf{soft cost formulation}
(\texttt{convert\_to\_binary} is not set). Binary conversion only occurs in the
metrics rollout when evaluating feasibility.
