\section{优化问题}

\begin{equation}
\min_{\mathbf{u} \in \mathbb{R}^{N \times d}} \;
\mathcal{L}(\mathbf{u}) =
J_{\text{pose}} + J_{\text{bound}} + J_{\text{reg}} + J_{\text{target}} + J_{\text{rel}}
+ C_{\text{scene}} + C_{\text{self}} + C_{\text{rel}}
\end{equation}

其中 $N = 16$ 为 B-spline knot 数（action horizon），$d = 14$ 为关节自由度（双臂各 7 DoF）。
$\mathbf{u} = [\mathbf{u}_1, \ldots, \mathbf{u}_N]^\top$ 为 B-spline 控制点序列，
由 LBFGS 优化器直接最小化加权目标函数。

所有代价项均为\textbf{软代价}（连续、可微）。碰撞等项在优化中使用软代价形式，
在指标评估（metrics rollout）中通过 \texttt{convert\_to\_binary} 切换为二值可行性判定。

\section{软代价 $\mathcal{L}_{\text{soft}}(\mathbf{u})$}

\subsection{末端位姿追踪 $J_{\text{pose}}$}

对每个末端执行器 frame $f \in \mathcal{F}$，使用 \textbf{axis-angle} 方法
（配置: \texttt{use\_lie\_group: false}）：

\begin{equation}
J_{\text{pose}} = \sum_{f \in \mathcal{F}} \sum_{k=1}^{N}
\Bigg[ \frac{w_p}{2} \bigl\lVert \mathbf{W}_p \odot (\mathbf{p}_{k,f} - \mathbf{p}_f^{\text{goal}}) \bigr\rVert^2
+ w_r \lVert\boldsymbol{\omega}_{k,f}\rVert^2 \Bigg]
\end{equation}

其中 $\mathbf{W}_p = \operatorname{diag}(\hat{w}_x^p, \hat{w}_y^p, \hat{w}_z^p) \in \mathbb{R}^{3 \times 3}$ 为逐轴位置权重矩阵，
$\odot$ 为逐元素乘法。$\boldsymbol{\omega}_{k,f} = \theta_{k,f} \cdot \mathbf{a}_{k,f}$ 为 axis-angle 表示，由相对旋转提取：
%
\begin{align*}
\mathbf{q}_{\Delta} &= \mathbf{q}_{k,f} \otimes \bigl(\mathbf{q}_f^{\text{goal}}\bigr)^{-1} \\
\mathbf{v} &= \mathbf{W}_r \odot \operatorname{vec}(\mathbf{q}_{\Delta}) \\
\theta &= 2 \operatorname{arctan2}\bigl(\lVert\mathbf{v}\rVert,\; |w|\bigr) \\
\mathbf{a} &= \frac{\mathbf{v}}{\lVert\mathbf{v}\rVert}
	\qquad (\lVert\mathbf{v}\rVert > \varepsilon \text{ 时}) \\
\boldsymbol{\omega}_{k,f} &= \theta \cdot \mathbf{a}
\end{align*}

梯度表达式：
%
\begin{equation}
\nabla_{\mathbf{p}} J_{\text{pose}} = w_p \cdot \mathbf{W}_p^2 \odot (\mathbf{p} - \mathbf{p}^{\text{goal}}),
\qquad
\nabla_{\boldsymbol{\omega}} J_{\text{pose}} = 2 w_r \boldsymbol{\omega}
\end{equation}

\begin{quote}
\textbf{注：} 当 \texttt{use\_lie\_group: true} 时，改用 Lie 群对数映射
$\log: SO(3) \to \mathfrak{so}(3)$，公式等价但 Jacobian 计算更精确。
\end{quote}

\subsection{关节空间 Bound 惩罚 $J_{\text{bound}}$}

对每个关节 $d$ 的每个状态量 $s \in \{\text{pos}, \text{vel}, \text{acc}, \text{jerk}, \text{torque}\}$，
当状态超出\textbf{激活距离收缩后的边界}时施加二次惩罚：

\begin{equation}
J_{\text{bound}} = \sum_{k=1}^{N} \sum_{d=1}^{d} \sum_{s}
\frac{w_b^s}{2} \Bigl(
\bigl[\max(0,\; x_{k,d}^s - \bar{u}_d^s)\bigr]^2
+ \bigl[\max(0,\; \bar{l}_d^s - x_{k,d}^s)\bigr]^2
\Bigr)
\end{equation}

其中收缩边界为（$\eta^s$ 为激活距离比例）：
%
\begin{equation}
\bar{l}_d^s = l_d^s + \eta^s (u_d^s - l_d^s), \qquad
\bar{u}_d^s = u_d^s - \eta^s (u_d^s - l_d^s)
\end{equation}

梯度（仅在越界时非零）：
%
\begin{equation}
\nabla_{x} J_{\text{bound}} =
\begin{cases}
w_b^s (x - \bar{u}_d^s) & x > \bar{u}_d^s \\
w_b^s (x - \bar{l}_d^s) & x < \bar{l}_d^s \\
0 & \text{otherwise}
\end{cases}
\end{equation}

\texttt{retime\_weights: false} 表示权重不随 $dt$ 缩放。

\subsection{L2 平滑正则化 $J_{\text{reg}}$}

始终激活的 Tikhonov 正则化，与 bound 惩罚独立：

\begin{equation}
J_{\text{reg}} = \sum_{k=1}^{N} \sum_{d=1}^{d}
\frac{1}{2} \Bigl[ \lambda_v \dot{q}_{k,d}^2 + \lambda_a \ddot{q}_{k,d}^2 + \lambda_j \dddot{q}_{k,d}^2 \Bigr]
\end{equation}

\texttt{retime\_regularization\_weights: true} 表示权重按 $dt^n$ 缩放（$\lambda_a$ 缩放 $dt^2$），使正则化近似
时间积分 $\int \lambda_a \lVert\ddot{\mathbf{q}}(t)\rVert^2 dt$，对不同 $dt$ 保持一致的平滑效果。

\subsection{关节目标距离 $J_{\text{target}}$}

将轨迹拉向默认关节配置（retract pose），终端权重远大于非终端：

\begin{equation}
J_{\text{target}} = \sum_{k=1}^{N} \alpha_k \, w_t \sum_{d=1}^{d} (q_{k,d} - q_d^{\text{target}})^2
\end{equation}
%
\begin{equation}
\alpha_k = \begin{cases}
1.0  & k = N \quad \text{(terminal knot)} \\
0.05 & k < N \quad \text{(non-terminal)}
\end{cases}
\end{equation}

梯度：$\nabla_{q_{k,d}} J_{\text{target}} = 2 \alpha_k w_t (q_{k,d} - q_d^{\text{target}})$。

\subsection{相对位姿协同 $J_{\text{rel}}$（运行时可选）}

双臂协同搬运时启用，约束 secondary 末端相对于 primary 末端的位姿：

\begin{equation}
J_{\text{rel}} = \sum_{k=1}^{N}
\Bigg[ \frac{w_{\text{rel},p}}{2} \bigl\lVert \Delta\mathbf{p}_k - \Delta\mathbf{p}^{\text{target}} \bigr\rVert^2
+ w_{\text{rel},r} \bigl\lVert \boldsymbol{\omega}_k^{\text{rel}} \bigr\rVert^2 \Bigg]
\end{equation}

相对位姿在 primary 局部坐标系中计算（$\mathbf{q}^*$ 为四元数共轭）：
%
\begin{align}
\Delta\mathbf{p}_k &= \mathbf{q}_{k,\text{primary}}^* \odot \bigl( \mathbf{p}_{k,\text{secondary}} - \mathbf{p}_{k,\text{primary}} \bigr) \\
\Delta\mathbf{q}_k &= \mathbf{q}_{k,\text{primary}}^* \otimes \mathbf{q}_{k,\text{secondary}}
\end{align}

位置梯度从局部坐标系传播至世界坐标系：
%
\begin{equation}
\nabla_{\mathbf{p}_1} = -\mathbf{R}(\mathbf{q}_1) \cdot \nabla_{\Delta\mathbf{p}}, \qquad
\nabla_{\mathbf{p}_2} = \mathbf{R}(\mathbf{q}_1) \cdot \nabla_{\Delta\mathbf{p}}
\end{equation}

\section{硬约束 / 碰撞代价}

\subsection{场景碰撞代价 $C_{\text{scene}}$}

对每个 robot sphere $i$ 在每个时间步，使用 \textbf{平滑激活函数}
（C$^1$ 连续，quadratic-linear transition）：

\begin{equation}
C_{\text{scene}} = w_{\text{col}} \sum_{k=1}^{N} \sum_{i=1}^{N_s} f_{\eta}\!\bigl(d_{k,i}^{\text{pen}}\bigr)
\end{equation}

穿透深度为 $d_{k,i}^{\text{pen}} = r_i + \eta - \operatorname{sdf}(\mathbf{c}_i)$
（$\operatorname{sdf}$ 在障碍物内部为负）。平滑激活函数：
%
\begin{equation}
f_{\eta}(d) = \begin{cases}
0                        & d \le 0 \\[4pt]
\dfrac{d^2}{2\eta}       & 0 < d \le \eta \quad \text{(quadratic region)} \\[8pt]
d - \dfrac{\eta}{2}      & d > \eta \quad \text{(linear region)}
\end{cases}
\end{equation}

该函数在 $d = 0$ 和 $d = \eta$ 处 C$^1$ 连续：
%
\begin{equation}
f_{\eta}(0) = 0,\; f_{\eta}'(0) = 0; \qquad
f_{\eta}(\eta) = \frac{\eta}{2},\; f_{\eta}'(\eta) = 1
\end{equation}

梯度分段：
$d \le 0 \Rightarrow 0$；
$0 < d \le \eta \Rightarrow d / \eta$；
$d > \eta \Rightarrow 1$。

Sweep 模式 (\texttt{use\_sweep: true}, \texttt{use\_speed\_metric: true}) 额外考虑 sphere
在 $dt$ 内的运动速度，动态缩放激活距离以补偿离散时间步的碰撞检测盲区。

\subsection{自碰撞代价 $C_{\text{self}}$}

对所有预定义的 sphere pair $(i,j)$，使用平方距离差惩罚，取最大穿透 pair：

\begin{equation}
C_{\text{self}} = w_{\text{self}} \sum_{k=1}^{N}
\max_{(i,j)} \Bigl[ \frac{1}{2} \bigl[ \max\bigl(0,\;
(r_i + r_j + \delta_{ij})^2 - \lVert\mathbf{c}_{k,i} - \mathbf{c}_{k,j}\rVert^2 \bigr) \bigr] \Bigr]
\end{equation}

其中 $\delta_{ij}$ 为 pair-specific padding（碰撞缓冲）。梯度（仅在最大穿透 pair 上计算，稀疏梯度）：
%
\begin{equation}
\nabla_{\mathbf{c}_i} C_{\text{self}} = -w_{\text{self}} (\mathbf{c}_i - \mathbf{c}_j), \qquad
\nabla_{\mathbf{c}_j} C_{\text{self}} = w_{\text{self}} (\mathbf{c}_i - \mathbf{c}_j)
\end{equation}

\subsection{相对位姿硬约束 $C_{\text{rel}}$（运行时可选）}

通过 \texttt{convert\_to\_binary: True} 将相对位姿代价转为二值约束，带容差：

\begin{equation}
C_{\text{rel}} = w_{\text{rel}}^{\text{hard}} \sum_{k=1}^{N}
\Bigl[ \max\bigl(0,\; \lVert\Delta\mathbf{p}_k - \Delta\mathbf{p}^{\text{target}}\rVert - \delta_p\bigr)
+ \max\bigl(0,\; \theta_k^{\text{rel}} - \delta_r\bigr) \Bigr]
\end{equation}

\section{Cost / Constraint / Convergence 分类}

在 cuRobo 的 rollout 架构中，代价项分为三类：

\begin{center}
\begin{tabular}{@{}llll@{}}
\toprule
类别 & 配置字段 & 行为 & 示例 \\
\midrule
\textbf{Cost}       & \texttt{cost\_cfg}       & 平滑惩罚，被优化器最小化 & 位姿误差、CSpace 距离、平滑项 \\
\textbf{Constraint} & \texttt{constraint\_cfg} & 二值可行性判定 (\texttt{convert\_to\_binary: True}) & 碰撞、关节限位 \\
\textbf{Convergence}& \texttt{convergence\_cfg} & 容差检查，决定终止条件 & 位置/姿态容差 \\
\bottomrule
\end{tabular}
\end{center}

成功条件 = $\text{feasible} \land \text{converged}$。
Feasibility 要求所有 constraint 值 $\le 0$。
Convergence 要求所有 tolerance metric 低于其阈值。

\textbf{注意：} 在 MPC 优化过程中，碰撞代价在 \texttt{cost\_cfg} 和 \texttt{constraint\_cfg} 中\textbf{均使用软代价形式}
（未开启 \texttt{convert\_to\_binary}）。二值化仅在 metrics rollout 评估可行性时启用。
