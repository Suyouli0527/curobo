\section{Optimization Problem}

\begin{equation}
\min_{\mathbf{u} \in \mathbb{R}^{N \times d}} \;
\mathcal{L}(\mathbf{u}) =
J_{\text{pose}} + J_{\text{bound}} + J_{\text{reg}} + J_{\text{target}} + J_{\text{rel}}
+ C_{\text{scene}} + C_{\text{self}} + C_{\text{rel}}
\end{equation}

where $N = 16$ is the number of B-spline knots (action horizon),
$d = 14$ is the total joint DOF (7 per arm).
$\mathbf{u} = [\mathbf{u}_1, \ldots, \mathbf{u}_N]^\top \in \mathbb{R}^{N \times d}$
is the sequence of B-spline control points, directly optimized by LBFGS.

All terms are \textbf{soft costs} (continuous, differentiable) during optimization.
Collision costs are evaluated as hard binary constraints only in the metrics rollout
via \texttt{convert\_to\_binary}.

\section{Soft Costs}

\subsection{End-Effector Pose Tracking $J_{\text{pose}}$}

For each tool frame $f \in \mathcal{F}$, using the \textbf{axis-angle} method
(\texttt{use\_lie\_group: false}):

\begin{equation}
J_{\text{pose}} = \sum_{f \in \mathcal{F}} \sum_{k=1}^{N}
\Bigg[ \frac{w_p}{2} \bigl\lVert \mathbf{W}_p \odot (\mathbf{p}_{k,f} - \mathbf{p}_f^{\text{goal}}) \bigr\rVert^2
+ w_r \lVert\boldsymbol{\omega}_{k,f}\rVert^2 \Bigg]
\end{equation}

where $\mathbf{W}_p = \operatorname{diag}(\hat{w}_x^p, \hat{w}_y^p, \hat{w}_z^p) \in \mathbb{R}^{3 \times 3}$
is the per-axis position weight matrix, $\odot$ denotes element-wise multiplication.
$\boldsymbol{\omega}_{k,f} = \theta_{k,f} \cdot \mathbf{a}_{k,f}$ is the axis-angle
representation extracted from the relative rotation:
%
\begin{align*}
\mathbf{q}_{\Delta} &= \mathbf{q}_{k,f} \otimes \bigl(\mathbf{q}_f^{\text{goal}}\bigr)^{-1} \\
\mathbf{v} &= \mathbf{W}_r \odot \operatorname{vec}(\mathbf{q}_{\Delta}) \\
\theta &= 2 \operatorname{arctan2}\bigl(\lVert\mathbf{v}\rVert,\; |w|\bigr) \\
\mathbf{a} &= \frac{\mathbf{v}}{\lVert\mathbf{v}\rVert}
	\qquad (\lVert\mathbf{v}\rVert > \varepsilon) \\
\boldsymbol{\omega}_{k,f} &= \theta \cdot \mathbf{a}
\end{align*}

Gradient expressions:
%
\begin{equation}
\nabla_{\mathbf{p}} J_{\text{pose}} = w_p \cdot \mathbf{W}_p^2 \odot (\mathbf{p} - \mathbf{p}^{\text{goal}}),
\qquad
\nabla_{\boldsymbol{\omega}} J_{\text{pose}} = 2 w_r \boldsymbol{\omega}
\end{equation}

\begin{quote}
\textbf{Note:} When \texttt{use\_lie\_group: true}, the Lie group logarithmic map
$\log: SO(3) \to \mathfrak{so}(3)$ is used instead, giving equivalent results with
more accurate Jacobian computation.
\end{quote}

\subsection{Joint-Space Bound Penalty $J_{\text{bound}}$}

For each joint $d$ and each state quantity
$s \in \{\text{pos}, \text{vel}, \text{acc}, \text{jerk}, \text{torque}\}$,
a quadratic penalty is applied when the state exceeds the \textbf{activation-shrunken bounds}:

\begin{equation}
J_{\text{bound}} = \sum_{k=1}^{N} \sum_{d=1}^{d} \sum_{s}
\frac{w_b^s}{2} \Bigl(
\bigl[\max(0,\; x_{k,d}^s - \bar{u}_d^s)\bigr]^2
+ \bigl[\max(0,\; \bar{l}_d^s - x_{k,d}^s)\bigr]^2
\Bigr)
\end{equation}

The shrunken bounds are ($\eta^s$ is the activation distance ratio):
%
\begin{equation}
\bar{l}_d^s = l_d^s + \eta^s (u_d^s - l_d^s), \qquad
\bar{u}_d^s = u_d^s - \eta^s (u_d^s - l_d^s)
\end{equation}

Gradient (non-zero only when bounds are violated):
%
\begin{equation}
\nabla_{x} J_{\text{bound}} =
\begin{cases}
w_b^s (x - \bar{u}_d^s) & x > \bar{u}_d^s \\
w_b^s (x - \bar{l}_d^s) & x < \bar{l}_d^s \\
0 & \text{otherwise}
\end{cases}
\end{equation}

\texttt{retime\_weights: false} means weights are not scaled by $dt$.

\subsection{L2 Smoothness Regularization $J_{\text{reg}}$}

Always-active Tikhonov regularization, independent of the bound penalty:

\begin{equation}
J_{\text{reg}} = \sum_{k=1}^{N} \sum_{d=1}^{d}
\frac{1}{2} \Bigl[ \lambda_v \dot{q}_{k,d}^2 + \lambda_a \ddot{q}_{k,d}^2 + \lambda_j \dddot{q}_{k,d}^2 \Bigr]
\end{equation}

\texttt{retime\_regularization\_weights: true} means weights are scaled by $dt^n$
(e.g.\ $\lambda_a$ scaled by $dt^2$), making the regularization approximate the
time integral $\int \lambda_a \lVert\ddot{\mathbf{q}}(t)\rVert^2 dt$ and remain
consistent across different $dt$ values.

\subsection{Joint Target Distance $J_{\text{target}}$}

Pulls the trajectory toward the default joint configuration (retract pose),
with terminal weights much larger than non-terminal:

\begin{equation}
J_{\text{target}} = \sum_{k=1}^{N} \alpha_k \, w_t \sum_{d=1}^{d} (q_{k,d} - q_d^{\text{target}})^2
\end{equation}
%
\begin{equation}
\alpha_k = \begin{cases}
1.0  & k = N \quad \text{(terminal knot)} \\
0.05 & k < N \quad \text{(non-terminal)}
\end{cases}
\end{equation}

Gradient: $\nabla_{q_{k,d}} J_{\text{target}} = 2 \alpha_k w_t (q_{k,d} - q_d^{\text{target}})$.

\subsection{Relative Pose Coordination $J_{\text{rel}}$ (runtime toggle)}

Enabled during dual-arm collaborative transport. Constrains the secondary end-effector's
pose relative to the primary:

\begin{equation}
J_{\text{rel}} = \sum_{k=1}^{N}
\Bigg[ \frac{w_{\text{rel},p}}{2} \bigl\lVert \Delta\mathbf{p}_k - \Delta\mathbf{p}^{\text{target}} \bigr\rVert^2
+ w_{\text{rel},r} \bigl\lVert \boldsymbol{\omega}_k^{\text{rel}} \bigr\rVert^2 \Bigg]
\end{equation}

The relative pose is computed in the primary's local frame ($\mathbf{q}^*$ = quaternion conjugate):
%
\begin{align}
\Delta\mathbf{p}_k &= \mathbf{q}_{k,\text{primary}}^* \odot \bigl( \mathbf{p}_{k,\text{secondary}} - \mathbf{p}_{k,\text{primary}} \bigr) \\
\Delta\mathbf{q}_k &= \mathbf{q}_{k,\text{primary}}^* \otimes \mathbf{q}_{k,\text{secondary}}
\end{align}

Position gradient propagation to world frame:
%
\begin{equation}
\nabla_{\mathbf{p}_1} = -\mathbf{R}(\mathbf{q}_1) \cdot \nabla_{\Delta\mathbf{p}}, \qquad
\nabla_{\mathbf{p}_2} = \mathbf{R}(\mathbf{q}_1) \cdot \nabla_{\Delta\mathbf{p}}
\end{equation}

\section{Collision Costs}

\subsection{Scene Collision Cost $C_{\text{scene}}$}

For each robot sphere $i$ at each timestep, using a \textbf{smooth activation function}
($C^1$-continuous, quadratic-linear transition):

\begin{equation}
C_{\text{scene}} = w_{\text{col}} \sum_{k=1}^{N} \sum_{i=1}^{N_s} f_{\eta}\!\bigl(d_{k,i}^{\text{pen}}\bigr)
\end{equation}

The penetration depth is
$d_{k,i}^{\text{pen}} = r_i + \eta - \operatorname{sdf}(\mathbf{c}_i)$
($\operatorname{sdf}$ is negative inside obstacles).
The smooth activation function (source: \texttt{wp\_collision\_common.py:12-38}):

\begin{equation}
f_{\eta}(d) = \begin{cases}
0                        & d \le 0 \\[4pt]
\dfrac{d^2}{2\eta}       & 0 < d \le \eta \quad \text{(quadratic region)} \\[8pt]
d - \dfrac{\eta}{2}      & d > \eta \quad \text{(linear region)}
\end{cases}
\end{equation}

The function is $C^1$-continuous at $d = 0$ and $d = \eta$:
%
\begin{equation}
f_{\eta}(0) = 0,\; f_{\eta}'(0) = 0; \qquad
f_{\eta}(\eta) = \frac{\eta}{2},\; f_{\eta}'(\eta) = 1
\end{equation}

Gradient:
$d \le 0 \Rightarrow 0$;
$0 < d \le \eta \Rightarrow d / \eta$;
$d > \eta \Rightarrow 1$.

Sweep mode (\texttt{use\_sweep: true}, \texttt{use\_speed\_metric: true}) additionally
accounts for sphere velocity over $dt$, dynamically scaling the activation distance to
compensate for the blind spot between discrete timesteps.

\subsection{Self-Collision Cost $C_{\text{self}}$}

For all predefined sphere pairs $(i,j)$, using a squared distance-difference penalty
on the maximally penetrating pair:

\begin{equation}
C_{\text{self}} = w_{\text{self}} \sum_{k=1}^{N}
\max_{(i,j)} \Bigl[ \frac{1}{2} \bigl[ \max\bigl(0,\;
(r_i + r_j + \delta_{ij})^2 - \lVert\mathbf{c}_{k,i} - \mathbf{c}_{k,j}\rVert^2 \bigr) \bigr] \Bigr]
\end{equation}

where $\delta_{ij}$ is the pair-specific padding.
Gradient (sparse, computed only for the max-penetration pair):
%
\begin{equation}
\nabla_{\mathbf{c}_i} C_{\text{self}} = -w_{\text{self}} (\mathbf{c}_i - \mathbf{c}_j), \qquad
\nabla_{\mathbf{c}_j} C_{\text{self}} = w_{\text{self}} (\mathbf{c}_i - \mathbf{c}_j)
\end{equation}

\subsection{Relative Pose Hard Constraint $C_{\text{rel}}$ (runtime toggle)}

Uses \texttt{convert\_to\_binary: True} to turn the relative pose cost into a binary
constraint with tolerances $\delta_p, \delta_r$:

\begin{equation}
C_{\text{rel}} = w_{\text{rel}}^{\text{hard}} \sum_{k=1}^{N}
\Bigl[ \max\bigl(0,\; \lVert\Delta\mathbf{p}_k - \Delta\mathbf{p}^{\text{target}}\rVert - \delta_p\bigr)
+ \max\bigl(0,\; \theta_k^{\text{rel}} - \delta_r\bigr) \Bigr]
\end{equation}

\section{Cost / Constraint / Convergence Taxonomy}

In cuRobo's rollout architecture, cost terms are classified into three categories:

\begin{center}
\begin{tabular}{@{}llll@{}}
\toprule
Category & Config Field & Behavior & Examples \\
\midrule
\textbf{Cost}       & \texttt{cost\_cfg}       & Smooth penalty, minimized by optimizer & Pose error, cspace distance, smoothness \\
\textbf{Constraint} & \texttt{constraint\_cfg} & Binary feasibility (\texttt{convert\_to\_binary: True}) & Collision, joint limits \\
\textbf{Convergence}& \texttt{convergence\_cfg} & Tolerance check for termination & Position/orientation tolerance \\
\bottomrule
\end{tabular}
\end{center}

Success condition = $\text{feasible} \land \text{converged}$.
Feasibility requires all constraint values $\le 0$.
Convergence requires all tolerance metrics below their thresholds.

\textbf{Important:} During MPC optimization, collision costs in both \texttt{cost\_cfg}
and \texttt{constraint\_cfg} use \textbf{soft cost formulation}
(\texttt{convert\_to\_binary} is not set). Binary conversion only occurs in the
metrics rollout when evaluating feasibility.
